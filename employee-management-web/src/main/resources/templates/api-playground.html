Terminal / Postman cURL Examples

        
# 1. Fetch all employees in JSON
curl -X GET http://localhost:8080/api/employees

# 2. Search employees
curl -X GET "http://localhost:8080/api/employees?search=Alice"

# 3. Create employee via POST JSON
curl -X POST http://localhost:8080/api/employees \
     -H "Content-Type: application/json" \
     -d '{"name":"Sam Taylor","email":"sam@example.com","salary":64000,"departmentId":1}'

# 4. Authenticate and get JWT token
curl -X POST http://localhost:8080/api/auth/login \
     -H "Content-Type: application/json" \
     -d '{"username":"hr","password":"1234"}'
